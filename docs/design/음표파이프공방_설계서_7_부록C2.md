### C.7 소리

```js
// ── 소리: 합성음만 쓴다. 모든 음을 오디오 클록에 미리 예약하고 한꺼번에 취소한다 ──
let ctx = null, master = null, noiseBuf = null, wave = null, run = null, raf = 0, valve = null, endedAt = -1e9, valved = false;
const live = new Set();
const makeWave = (c) => c.createPeriodicWave(new Float32Array(7), new Float32Array([0, 1, 0.42, 0.26, 0.1, 0.06, 0.03]));      // 관 소리: 배음을 조금 얹은 지속음
function audio() {
  try {
    if (!ctx) {
      const AC = window.AudioContext || window.webkitAudioContext;
      if (!AC) return null;
      ctx = new AC();
      master = ctx.createGain(); master.gain.value = 0.8; master.connect(ctx.destination);
      noiseBuf = ctx.createBuffer(1, ctx.sampleRate, ctx.sampleRate);
      const d = noiseBuf.getChannelData(0); for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1;
      wave = makeWave(ctx);
      ctx.addEventListener("statechange", syncSound);
    }
    if (ctx.state !== "running") { const p = ctx.resume(); if (p && p.catch) p.catch(() => {}); }
    syncSound();
    return ctx;
  } catch (e) { return null; }
}
function syncSound() {
  const on = !!ctx && ctx.state === "running";
  $("soundChip").hidden = on; $("hud").classList.toggle("nosound", !on);
  if (!on && run && !run.perf) abortRun();
}
const keep = (node) => { live.add(node); node.addEventListener("ended", () => live.delete(node)); };
const silence = () => { live.forEach((x) => { try { x.stop(); } catch (e) { /* 이미 끝남 */ } }); live.clear(); };
function pipeTone(midi, when, len, attack, release) {                               // attack: fresh 새로 시작 · legato 이어서 시작. release: cut 끊김 · over 다음 음과 겹침 · dip 같은 음을 다시 내려고 잠깐 꺼짐
  const o = ctx.createOscillator(), g = ctx.createGain(), peak = 0.24, up = attack === "legato" ? 0.03 : 0.018, end = when + len;
  o.setPeriodicWave(wave); o.frequency.setValueAtTime(hz(midi), when);
  g.gain.setValueAtTime(0.0001, when); g.gain.linearRampToValueAtTime(peak, when + up);
  if (release === "over") { g.gain.setValueAtTime(peak, end); g.gain.linearRampToValueAtTime(0.0001, end + 0.03); }
  else if (release === "dip") { g.gain.setValueAtTime(peak, Math.max(when + up, end - 0.07)); g.gain.linearRampToValueAtTime(0.0001, end - 0.012); }
  else { g.gain.setValueAtTime(peak, Math.max(when + up, end - 0.04)); g.gain.linearRampToValueAtTime(0.0001, end); }
  o.connect(g); g.connect(master); o.start(when); o.stop(end + 0.06); keep(o);
}
function burst(when, dur, freq, q, peak) {
  const s = ctx.createBufferSource(), f = ctx.createBiquadFilter(), g = ctx.createGain();
  s.buffer = noiseBuf; f.type = "bandpass"; f.frequency.value = freq; f.Q.value = q;
  g.gain.setValueAtTime(peak, when); g.gain.exponentialRampToValueAtTime(0.0001, when + dur);
  s.connect(f); f.connect(g); g.connect(master); s.start(when); s.stop(when + dur + 0.02); keep(s);
  return s;
}
const drop = (when, fresh) => burst(when, 0.035, 1900, 1.4, fresh ? 0.22 : 0.08);    // 방울 소리
const tick = (when, strong) => burst(when, 0.035, strong ? 2600 : 1700, 6, strong ? 0.6 : 0.35);
const lat = () => (ctx ? ctx.outputLatency || ctx.baseLatency || 0 : 0);
const clock = () => (run && run.perf ? performance.now() / 1000 : ctx.currentTime - lat());       // 지금 들리는 시각
```

### C.8 재생, 밸브, 끝내기, 중단, 입력

`frame`의 첫 부분이 화면 시계로의 전환과 정지 감시다. `startValve`가 누르는 동안의 발진기와 주파수 예약이다. `swallow`와 `keyHeld`는 밸브 도중에 누른 손가락과 키를 뗄 때 생기는 `click`을 버린다.

```js
// ── 재생 다섯 가지: trial(시운전) · hear(의뢰 듣기, 소리만) · listen(납품 듣기) · hand(밸브) · machine(기계 밸브) ──
function begin(kind) {
  if (kind === "hand" && performance.now() - endedAt < 600) return;                  // 끝난 직후의 손짓으로 다시 시작하지 않는다
  stopRun(false);
  const a = audio(), job = kind === "trial" ? shownJob() : truthNow(), bpm = S.slow ? 50 : 60, beat = 60 / bpm, total = num(totalBeats(job));
  const valving = kind === "hand" || kind === "machine", countIn = valving ? (valved ? 2 : 4) : 0;
  const t0 = (a ? a.currentTime : performance.now() / 1000) + 0.12;
  const r = { kind, job, beat, total, countIn, perf: !a, t0, tb0: t0 + countIn * beat, holds: [], down: null, ctx0: a ? a.currentTime : 0, p0: performance.now(), lastNow: null, lastAt: 0 };
  r.end = r.tb0 + total * beat + 0.3;
  if (kind === "machine") r.plan = pressPlan(job).map(([x, y]) => [num(x), num(y)]);
  if (a) {
    if (valving) {
      for (let k = 0; k < countIn; k++) tick(t0 + k * beat, k === 0);
      for (let k = 0; k <= Math.ceil(total - 1e-9); k++) tick(r.tb0 + k * beat, k % job.meter[0] === 0);
    }
    if (kind !== "hand") schedule(job, bpm).forEach((x) => { pipeTone(x.midi, r.tb0 + x.t, x.len, x.attack, x.release); drop(r.tb0 + x.t, x.attack === "fresh"); });
  }
  run = r;
  if (valving) valved = true;
  resetPaint();
  setState(kind === "hear" ? { hearing: true, sub: "ready" } : kind === "listen" ? { sub: "listen", hearing: false } : kind === "trial" ? { sub: "run", hearing: false } : { sub: "countin", label: "1", hearing: false, marks: null, msg: { tone: "", text: "" } });
  raf = requestAnimationFrame(frame);
}
function resetPaint() {
  document.querySelectorAll("#pipes .water.auto").forEach((e) => e.setAttribute("width", "0"));
  document.querySelectorAll("#pipes .riserWater").forEach((e) => e.classList.remove("on"));
  document.querySelectorAll("#pipes .bell").forEach((e) => e.classList.remove("rung"));
  const hw = $("handWater"); if (hw) hw.innerHTML = "";
  $("pad").classList.remove("down");
}
function paintAuto(b) {
  document.querySelectorAll("#pipes .water.auto").forEach((e, i) => { const w = P.waters[i]; e.setAttribute("width", String(Math.max(0, Math.min(1, (b - w.b0) / (w.b1 - w.b0))) * w.w)); });
  document.querySelectorAll("#pipes .riserWater").forEach((e, i) => e.classList.toggle("on", b >= P.risers[i].b));
  document.querySelectorAll("#pipes .bell").forEach((e, i) => { if (b >= P.bells[i].b && !e.classList.contains("rung")) e.classList.add("rung"); });
}
function paintHand(b, r) {                                                           // 누른 구간에만 물이 찬다
  const g = P.g, all = r.down === null ? r.holds : r.holds.concat([[r.down, Math.min(b, r.total)]]);
  let h = "";
  P.waters.forEach((w) => all.forEach(([a, c]) => {
    const p = Math.max(a, w.b0), q = Math.min(c, w.b1);
    if (q > p) { const x = g.X(p) + (p === w.b0 ? 3.5 : 0), x2 = g.X(q) - (q === w.b1 ? 3.5 : 0); if (x2 > x) h += `<rect class="water" x="${x}" y="${w.y}" width="${x2 - x}" height="${TH - 6}" rx="4"/>`; }
  }));
  $("handWater").innerHTML = h;
  document.querySelectorAll("#pipes .riserWater").forEach((e, i) => e.classList.toggle("on", all.some(([a, c]) => a <= P.risers[i].b - 1 / 16 && c >= P.risers[i].b + 1 / 16)));
  document.querySelectorAll("#pipes .bell").forEach((e, i) => { if (!e.classList.contains("rung") && all.some(([a, c]) => a <= P.bells[i].b + 0.25 && c > P.bells[i].b + 0.02)) e.classList.add("rung"); });
}
function frame() {
  const r = run;
  if (!r) return;
  if (!r.perf && performance.now() - r.p0 > 1200 && ctx.currentTime === r.ctx0) {      // 소리가 거부되면 화면 시계로 이어 간다
    const shift = performance.now() / 1000 + 0.1 - r.t0;
    r.perf = true; r.t0 += shift; r.tb0 += shift; r.end += shift; silence();
  }
  const now = clock(), b = (now - r.tb0) / r.beat;
  if (!r.perf) {                                                                      // 시작한 뒤에 소리 시계가 0.8초 넘게 서 있으면 중단한다
    if (now !== r.lastNow) { r.lastNow = now; r.lastAt = performance.now(); }
    else if (ctx.currentTime !== r.ctx0 && performance.now() - r.lastAt > 800) { abortRun(); return; }
  }
  if (r.kind !== "hear") {
    const g = P.g, x = g.X(Math.max(0, Math.min(b, r.total))), head = $("head"), sh = $("scoreHead");
    if (head) { head.setAttribute("x1", x); head.setAttribute("x2", x); }
    if (sh) { const sx = Math.min(g.W - 6, x + 13); sh.setAttribute("x1", sx); sh.setAttribute("x2", sx); }
    if (r.kind === "hand" || r.kind === "machine") {
      if (b < 0) { const lbl = String(Math.min(r.countIn, Math.max(1, r.countIn - Math.ceil(-b) + 1))); if (S.sub !== "countin" || S.label !== lbl) setState({ sub: "countin", label: lbl }); }
      else if (S.sub !== "run") setState({ sub: "run", label: "" });
    }
    if (r.kind === "hand") paintHand(b, r);
    else { paintAuto(b); if (r.kind === "machine") $("pad").classList.toggle("down", r.plan.some(([a, c]) => b >= a && b < c)); }
  }
  if (now >= r.end) { finish(); return; }
  raf = requestAnimationFrame(frame);
}
function stopValve() {
  if (!valve) return;
  const t = ctx.currentTime;
  try { valve.g.gain.cancelScheduledValues(t); valve.g.gain.setTargetAtTime(0.0001, t, 0.012); valve.o.stop(t + 0.09); } catch (e) { /* 이미 끝남 */ }
  valve.extras.forEach((x) => { try { x.stop(); } catch (e) { /* 이미 끝남 */ } });
  valve = null;
}
function startValve(r) {                                                             // 누르는 동안만 소리가 난다. 높이는 물길이 정한다
  const job = r.job, st = starts(job).map(num), t = ctx.currentTime, when = Math.max(t, r.tb0), endT = r.tb0 + r.total * r.beat;
  if (when >= endT) return;
  const bs = (when - r.tb0) / r.beat;
  let i0 = 0; st.forEach((s, i) => { if (s <= bs + 1e-9) i0 = i; });
  const o = ctx.createOscillator(), g = ctx.createGain(), extras = [];
  o.setPeriodicWave(wave); o.frequency.setValueAtTime(hz(midiOf(job.notes[i0].p)), when);
  for (let i = i0 + 1; i < st.length; i++) {
    const T = r.tb0 + st[i] * r.beat;
    o.frequency.setValueAtTime(hz(midiOf(job.notes[i].p)), T);
    if (!samePitch(job.notes[i].p, job.notes[i - 1].p)) extras.push(drop(T, false));
  }
  g.gain.setValueAtTime(0.0001, when); g.gain.linearRampToValueAtTime(0.24, when + 0.018);
  g.gain.setValueAtTime(0.24, endT - 0.04); g.gain.linearRampToValueAtTime(0.0001, endT);
  o.connect(g); g.connect(master); o.start(when); o.stop(endT + 0.06); keep(o);
  extras.push(drop(when, true));
  valve = { o, g, extras };
}
function padDown() {
  const r = run;
  if (!r || r.kind !== "hand" || r.down !== null) return;
  const b = (clock() - r.tb0) / r.beat;
  if (b >= r.total) return;
  r.down = b; $("pad").classList.add("down");
  if (ctx && !r.perf) startValve(r);
}
function padUp() {
  const r = run;
  if (!r || r.kind !== "hand") return;
  $("pad").classList.remove("down");
  if (r.down === null) return;
  const b = (clock() - r.tb0) / r.beat;
  if (b > 0) r.holds.push([r.down, Math.min(b, r.total)]);
  r.down = null; stopValve();
}
function finish() {
  const r = run;
  if (r.kind === "hand" && r.down !== null) { r.holds.push([r.down, r.total]); r.down = null; }
  stopRun(true);
  if (r.kind === "hear") { setState({ hearing: false }); return; }
  if (r.kind === "trial" || r.kind === "listen") { setState({ sub: "ready" }); return; }
  endedAt = performance.now();
  const best = (got) => ((RANK[got] || 0) > (RANK[S.valve] || 0) ? got : S.valve);
  if (r.kind === "machine") { setState({ sub: "result", label: "", valve: best("기계"), msg: { tone: "", text: "기계가 설계도대로 눌렀어요. 보고 따라 해 보세요." } }); return; }
  const j = judgeHold(r.holds, r.job);
  setState({ sub: "result", label: "", marks: j.joints, valve: best({ smooth: "매끈", drip: "한 곳 샘", leak: "여러 곳 샘" }[j.grade]), msg: valveText(j) });
}
function abortRun(why) {
  stopRun(false);
  setState({ sub: "ready", label: "", hearing: false, msg: { tone: "warn", text: why === "hidden" ? "화면을 떠나서 멈췄어요. 다시 시작해 주세요." : "소리가 멈춰서 중단했어요. 소리를 켜고 다시 시작해 주세요." } });
}
function stopRun(keepPaint) {
  cancelAnimationFrame(raf); raf = 0; stopValve(); silence(); run = null;
  if (!keepPaint) resetPaint(); else $("pad").classList.remove("down");
}
function stopAndIdle() { stopRun(false); setState({ sub: "ready", label: "", hearing: false }); }
```

### C.9 오디오 스파이(화면 점검용)

페이지 스크립트보다 먼저 실행한다(Playwright의 `addInitScript`). `WRAP`은 페이지가 뜬 뒤에 실행해 잡음 소리의 종류를 적는다.

```js
const SPY = () => {                                   // 오실레이터 생성을 가로채 주파수, 예약 시각, 취소 여부를 적는다
  const log = []; window.__audio = log;
  const AC = window.AudioContext; if (!AC) return;
  const mk = AC.prototype.createOscillator;
  AC.prototype.createOscillator = function () {
    const c = this, o = mk.call(this), e = { freqs: [], startAt: null, stopAt: null, calledAt: null, cancels: [] };
    const set = o.frequency.setValueAtTime.bind(o.frequency);
    o.frequency.setValueAtTime = (v, w) => { e.freqs.push([Math.round(v * 100) / 100, w]); return set(v, w); };
    const start = o.start.bind(o); o.start = (w) => { e.startAt = w; e.calledAt = c.currentTime; log.push(e); return start(w); };
    const stop = o.stop.bind(o); o.stop = (w) => { if (w === undefined) e.cancels.push(c.currentTime); else if (e.stopAt === null) e.stopAt = w; else e.restop = w; return stop(w); };
    return o;
  };
};
const WRAP = () => { window.__b = []; const b0 = burst; burst = function (when, dur, freq, q, peak) { window.__b.push({ when, freq, peak }); return b0(when, dur, freq, q, peak); }; };   // 잡음 소리(방울·딸깍)를 종류별로 적는다
```

- 재생 뒤 `window.__audio`의 주파수와 `startAt`의 차이가 2.2절의 소리 표와 같은지 본다(AU-01).
- `cancels`가 비어 있지 않으면 `stop()`이 인자 없이 불린 음이다. 그 시각이 새 재생의 `calledAt`보다 이르거나 같아야 한다(AU-02).
- 주파수를 `setValueAtTime`으로 넣어야 기록된다. 제품 코드도 그렇게 쓴다.
- 자동 재생이 막힌 기기는 `AudioContext`를 감싸 `state`를 `suspended`로, `currentTime`을 0으로 돌려주고 `resume()`이 끝나지 않게 해서 흉내 냈다(AU-08).
